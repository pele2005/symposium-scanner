// ---------- อ่านค่าตั้งค่า ----------
function getSettings_() {
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_SETTINGS);
  if (!sh) throw new Error('ไม่พบชีต "ตั้งค่า" – เลือกเมนู 1) ก่อน');
  var vals = sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues();
  var s = {};
  vals.forEach(function (r) { if (r[0]) s[String(r[0]).trim()] = r[1]; });
  s.PRIZE_LIMIT = Number(s.PRIZE_LIMIT) || 0;
  s.CODE_COUNT = Number(s.CODE_COUNT) || 0;
  s.SEND_EMAIL = String(s.SEND_EMAIL).toUpperCase() === 'TRUE';
  function ask3_(v, dflt) { var x = String(v === undefined || v === '' ? dflt : v).trim().toUpperCase(); return { ask: x !== 'FALSE', req: x === 'TRUE' }; }
  var lic = ask3_(s.ASK_LICENSE, 'TRUE'); s.ASK_LICENSE = lic.ask; s.LICENSE_REQUIRED = lic.req;
  var ph = ask3_(s.ASK_PHONE, 'TRUE'); s.ASK_PHONE = ph.ask; s.PHONE_REQUIRED = ph.req;
  var ln = ask3_(s.ASK_LINE, 'FALSE'); s.ASK_LINE = ln.ask; s.LINE_REQUIRED = ln.req;
  var ho = ask3_(s.ASK_HOSPITAL, 'FALSE'); s.ASK_HOSPITAL = ho.ask; s.HOSPITAL_REQUIRED = ho.req;
  var sp = ask3_(s.ASK_SPECIALTY, 'FALSE'); s.ASK_SPECIALTY = sp.ask; s.SPECIALTY_REQUIRED = sp.req;
  s.SPECIALTY_LIST = String(s.SPECIALTY_LIST || DEFAULT_SPECIALTIES).split(';').map(function (x) { return x.trim(); }).filter(String);
  s.START_TIME = toDate_(s.START_TIME);
  s.END_TIME = toDate_(s.END_TIME);
  s.SECRET_WORD = String(s.SECRET_WORD || '').trim();
  s.STAFF_PIN = String(s.STAFF_PIN || '').trim();
  s.EVENT_NAME = String(s.EVENT_NAME || 'Symposium');
  s.CODE_PREFIX = String(s.CODE_PREFIX || '').trim().toUpperCase();
  s.WEBAPP_URL = String(s.WEBAPP_URL || '').trim();
  s.REDEEM_NOTE = String(s.REDEEM_NOTE || '').trim();
  s.THANKS_EMAIL = String(s.THANKS_EMAIL).toUpperCase() === 'TRUE';
  s.THANKS_TEXT = String(s.THANKS_TEXT || '').trim();
  return s;
}

function setSetting_(key, value) {
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_SETTINGS);
  var keys = sh.getRange(1, 1, sh.getLastRow(), 1).getValues();
  for (var i = 0; i < keys.length; i++) {
    if (keys[i][0] === key) { sh.getRange(i + 1, 2).setValue(value); return; }
  }
  sh.appendRow([key, value, '']);
}

function toDate_(v) {
  if (!v) return null;
  if (v instanceof Date) return v;
  // รองรับ "dd/mm/yyyy hh:mm" (ปี ค.ศ. หรือ พ.ศ.)
  var m = String(v).trim().match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})(?:\s+(\d{1,2}):(\d{2}))?$/);
  if (m) {
    var y = Number(m[3]); if (y > 2400) y -= 543;
    return new Date(y, Number(m[2]) - 1, Number(m[1]), Number(m[4] || 0), Number(m[5] || 0));
  }
  var d = new Date(v);
  return isNaN(d.getTime()) ? null : d;
}

function getQuestions_() {
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_QUESTIONS);
  if (!sh || sh.getLastRow() < 2) return [];
  // ใช้ค่าที่แสดงบนชีต เพื่อให้ 50%, 1,000, วันที่ ตรงกับที่พิมพ์ไว้
  return sh.getRange(2, 1, sh.getLastRow() - 1, 4).getDisplayValues()
    .filter(function (r) { return String(r[0]).trim() !== ''; })
    .map(function (r) {
      var choices = String(r[1]).split(';').map(function (x) { return x.trim(); }).filter(String);
      var answers = String(r[2]).split(';').map(function (x) { return x.trim(); }).filter(String)
        .map(function (a) { return resolveChoice_(choices, a) || a; });
      var title = String(r[0]).trim();
      var kind = String(r[3] || '').trim();
      return {
        title: title,
        choices: choices,
        answers: answers,
        answer: answers.join(';'),
        graded: answers.length > 0,                       // ไม่มีเฉลย = แบบสำรวจ เก็บคำตอบอย่างเดียว
        multi: /หลาย|multi|checkbox/i.test(kind) || answers.length > 1 || /มากกว่า\s*1|หลายข้อ|multiple/i.test(title)
      };
    });
}

/** จับคู่คำตอบกับตัวเลือก: ตรงทั้งข้อความ หรือระบุแค่ตัวอักษรนำหน้า (A / ก / 1) */
function resolveChoice_(choices, token) {
  token = String(token || '').trim();
  if (!token) return '';
  for (var i = 0; i < choices.length; i++) if (choices[i] === token) return choices[i];
  var tl = token.toLowerCase().replace(/[\.\)]\s*$/, '');
  for (var j = 0; j < choices.length; j++) {
    var m = choices[j].match(/^([A-Za-z]|[ก-ฮ]|\d{1,2})\s*[\.\)]\s*/);
    if (m && m[1].toLowerCase() === tl) return choices[j];
  }
  for (var k = 0; k < choices.length; k++) if (choices[k].toLowerCase() === token.toLowerCase()) return choices[k];
  return '';
}
function sameSet_(a, b) {
  if (a.length !== b.length) return false;
  var m = {}; a.forEach(function (x) { m[x] = true; });
  return b.every(function (x) { return m[x]; });
}

function generateCodes() {
  var ui = SpreadsheetApp.getUi();
  var s = getSettings_();
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
  var existing = Math.max(0, sh.getLastRow() - 1);

  var toAdd = s.CODE_COUNT - existing;
  if (toAdd <= 0) { ui.alert('มีรหัสอยู่แล้ว ' + existing + ' รหัส (CODE_COUNT = ' + s.CODE_COUNT + ')\nถ้าต้องการเพิ่ม ให้เพิ่มค่า CODE_COUNT แล้วสั่งอีกครั้ง'); return; }

  var used = {};
  if (existing) sh.getRange(2, 1, existing, 1).getValues().forEach(function (r) { used[r[0]] = true; });

  var width = String(s.CODE_COUNT).length < 3 ? 3 : String(s.CODE_COUNT).length;
  var rows = [];
  for (var i = existing + 1; i <= s.CODE_COUNT; i++) {
    var code;
    do { code = s.CODE_PREFIX + pad_(i, width) + '-' + randomChars_(4); } while (used[code]);
    used[code] = true;
    var blank = [code, ST_EMPTY]; while (blank.length < CODE_HEADERS.length) blank.push('');
    rows.push(blank);
  }
  ensureCodeCols_(sh);
  sh.getRange(existing + 2, 1, rows.length, CODE_HEADERS.length).setValues(rows);
  sh.getRange(2, C.PHONE, s.CODE_COUNT, 1).setNumberFormat('@');
  sh.getRange(2, C.LICENSE, s.CODE_COUNT, 1).setNumberFormat('@');
  sh.getRange(2, C.SCORE, s.CODE_COUNT, 1).setNumberFormat('@');
  updateCardLinks(true);
  ui.alert('สร้างรหัสเพิ่ม ' + rows.length + ' รหัส (รวม ' + s.CODE_COUNT + ')\n\nขั้นต่อไป: เมนู 4) สร้างลิงก์บัตร แล้วพิมพ์บัตร');
}

function pad_(n, w) { var s = String(n); while (s.length < w) s = '0' + s; return s; }
function randomChars_(n) {
  var out = '';
  for (var i = 0; i < n; i++) out += CODE_ALPHABET.charAt(Math.floor(Math.random() * CODE_ALPHABET.length));
  return out;
}

function updateCardLinks(silent) {
  var s = getSettings_();
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
  var n = sh.getLastRow() - 1;
  if (n < 1) return;
  var codes = sh.getRange(2, 1, n, 1).getValues();
  var links, mode;
  if (s.WEBAPP_URL) {
    var base = s.WEBAPP_URL.split('?')[0];
    // หมายเหตุ: ห้ามใช้ชื่อพารามิเตอร์ "c" – Google บล็อกทำให้เว็บแอปขึ้น "unable to open the file" (HTTP 400)
    links = codes.map(function (r) { return [base + '?code=' + encodeURIComponent(r[0])]; });
    mode = 'Web App (หน้าสถานะ)';
  } else {
    var tmpl = PropertiesService.getScriptProperties().getProperty('PREFILL_TEMPLATE') || '';
    if (!tmpl) {
      if (silent !== true) SpreadsheetApp.getUi().alert('ยังไม่มีฟอร์ม – เลือกเมนู 2) สร้าง Google Form ก่อน');
      return;
    }
    links = codes.map(function (r) { return [tmpl.replace('CODEPLACEHOLDER', encodeURIComponent(r[0]))]; });
    mode = 'Google Form (กรอกรหัสบัตรให้อัตโนมัติ)';
  }
  sh.getRange(2, C.LINK, n, 1).setValues(links);
  if (silent !== true) SpreadsheetApp.getUi().alert('สร้างลิงก์บัตร ' + n + ' ใบ เรียบร้อย\nรูปแบบลิงก์: ' + mode + '\n\nคัดลอกคอลัมน์ J (ลิงก์บัตร) ทั้งหมด ไปวางในไฟล์ "พิมพ์บัตร QR.html" เพื่อพิมพ์บัตร');
}

// ---------- ตรวจคำตอบ (trigger) ----------
function normCode_(v) { return String(v || '').toUpperCase().replace(/\s+/g, '').trim(); }
function normId_(v) { return String(v || '').toUpperCase().replace(/[\s\-\.]/g, ''); }
function normPhone_(v) { var d = String(v || '').replace(/\D/g, ''); if (d.indexOf('66') === 0 && d.length === 11) d = '0' + d.slice(2); return d; }
function normWord_(v) { return String(v || '').toLowerCase().replace(/\s+/g, ''); }

function onFormSubmit(e) {
  var lock = LockService.getScriptLock();
  // ถ้ามีคนส่งพร้อมกันมาก ๆ จะรอคิว; ถ้ารอไม่ไหว ตัวเก็บตก (sweepMissed) จะมาตรวจให้ภายหลัง
  if (!lock.tryLock(240000)) return;
  try {
    if (isLogged_(e.response.getId())) return;
    processResponse_(e.response);
  } finally {
    lock.releaseLock();
  }
}

/** ตรวจคำตอบที่ตกหล่น (รันอัตโนมัติทุก 5 นาที และสั่งจากเมนูได้) */
function sweepMissed() {
  var s = getSettings_();
  if (!s.FORM_ID) return 0;
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(60000)) return 0;
  var count = 0;
  try {
    var logged = loggedIds_();
    FormApp.openById(s.FORM_ID).getResponses().forEach(function (r) {
      if (!logged[r.getId()]) { processResponse_(r); count++; }
    });
  } finally {
    lock.releaseLock();
  }
  return count;
}
function sweepMissedMenu() {
  var n = sweepMissed();
  SpreadsheetApp.getUi().alert('ตรวจคำตอบที่ตกหล่นแล้ว: ' + n + ' รายการ');
}
function loggedIds_() {
  var log = SpreadsheetApp.getActive().getSheetByName(SH_LOG);
  var o = {};
  if (log.getLastRow() < 2) return o;
  log.getRange(2, 9, log.getLastRow() - 1, 1).getValues().forEach(function (r) { if (r[0]) o[r[0]] = true; });

}
function isLogged_(id) { return !!loggedIds_()[id]; }

function processResponse_(resp) {
  var s = getSettings_();
  var ids = JSON.parse(PropertiesService.getScriptProperties().getProperty('ITEM_IDS') || '{}');
  var ans = {};
  resp.getItemResponses().forEach(function (ir) { ans[ir.getItem().getId()] = ir.getResponse(); });

  var d = {
    code: normCode_(ans[ids.code]),
    name: String(ans[ids.name] || '').trim(),
    license: ids.license ? normId_(ans[ids.license]) : '',
    phone: ids.phone ? normPhone_(ans[ids.phone]) : '',
    line: ids.line ? String(ans[ids.line] || '').trim().replace(/^@/, '') : '',
    hospital: ids.hospital ? String(ans[ids.hospital] || '').trim() : '',
    specialty: ids.specialty ? String(ans[ids.specialty] || '').trim() : '',
    email: String((typeof resp.getRespondentEmail === 'function' && resp.getRespondentEmail()) || ans[ids.email] || '').trim(),
    secret: ids.secret ? ans[ids.secret] : '',
    time: resp.getTimestamp()
  };

  var score = 0;
  var graded = (ids.questions || []).filter(function (q) { return q.graded !== false && (q.answers ? q.answers.length : String(q.answer || '').length); });
  graded.forEach(function (q) {
    var got = ans[q.id];
    var picked = (Array.isArray(got) ? got : String(got || '').split(/\s*;\s*/)).map(function (x) { return String(x).trim(); }).filter(String);
    var want = q.answers || String(q.answer || '').split(';').map(function (x) { return x.trim(); }).filter(String);
    if (sameSet_(picked, want)) score++;   // ข้อเลือกหลายข้อต้องเลือกครบและไม่เกิน
  });
  var total = graded.length;
  var minScore = (s.MIN_SCORE === '' || s.MIN_SCORE === null || s.MIN_SCORE === undefined) ? total : Number(s.MIN_SCORE);
  d.score = total ? score + '/' + total : 'สำรวจ';

  var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
  ensureCodeCols_(sh);
  var n = Math.max(0, sh.getLastRow() - 1);
  var data = n ? sh.getRange(2, 1, n, CODE_HEADERS.length).getValues() : [];

  var reason = '';
  var rowIdx = -1;
  for (var i = 0; i < data.length; i++) if (data[i][C.CODE - 1] === d.code) { rowIdx = i; break; }

  if (s.START_TIME && d.time < s.START_TIME) reason = 'ส่งก่อนเวลาเปิดรับคำตอบ';
  else if (s.END_TIME && d.time > s.END_TIME) reason = 'ส่งหลังเวลาปิดรับคำตอบ';
  else if (rowIdx < 0) reason = 'ไม่พบรหัสบัตรนี้ในระบบ';
  else if (data[rowIdx][C.STATUS - 1] !== ST_EMPTY) reason = 'รหัสบัตรนี้ถูกใช้สิทธิ์ไปแล้ว';
  else if (s.SECRET_WORD && normWord_(d.secret) !== normWord_(s.SECRET_WORD)) reason = 'คำลับไม่ถูกต้อง';
  else {
    for (var j = 0; j < data.length; j++) {
      var st = data[j][C.STATUS - 1];
      if (st !== ST_PASSED && st !== ST_REDEEMED) continue;
      if (d.license && normId_(data[j][C.LICENSE - 1]) === d.license) { reason = 'เลขใบประกอบวิชาชีพนี้ได้รับสิทธิ์แล้ว (บัตร ' + data[j][0] + ')'; break; }
      if (d.phone && normPhone_(data[j][C.PHONE - 1]) === d.phone) { reason = 'เบอร์โทรนี้ได้รับสิทธิ์แล้ว (บัตร ' + data[j][0] + ')'; break; }
      if (d.line && normWord_(data[j][C.LINE - 1]) === normWord_(d.line)) { reason = 'LINE ID นี้ได้รับสิทธิ์แล้ว (บัตร ' + data[j][0] + ')'; break; }
      if (d.email && normWord_(data[j][C.EMAIL - 1]) === normWord_(d.email)) { reason = 'อีเมลนี้ได้รับสิทธิ์แล้ว (บัตร ' + data[j][0] + ')'; break; }
    }
  }
  if (!reason && score < minScore) reason = 'ตอบถูก ' + score + ' จาก ' + total + ' ข้อ (ต้องถูกอย่างน้อย ' + minScore + ' ข้อ) – ตอบใหม่ได้';
  if (!reason) {
    var cnt = countStatus_();
    if (cnt.passed + cnt.redeemed >= s.PRIZE_LIMIT) reason = 'ของที่ระลึกหมดแล้ว';
  }

  var log = SpreadsheetApp.getActive().getSheetByName(SH_LOG);
  if (!reason) {
    var r = rowIdx + 2;
    sh.getRange(r, C.SCORE).setNumberFormat('@');
    sh.getRange(r, C.STATUS, 1, 7).setValues([[ST_PASSED, d.name, d.license, d.phone, d.email, d.score, d.time]]);
    sh.getRange(r, C.LINE, 1, 3).setValues([[d.line, d.hospital, d.specialty]]);
    log.appendRow([d.time, d.code, d.name, d.license, "'" + d.phone, 'ผ่าน', '', d.score, resp.getId(), d.line, d.hospital, d.specialty]);
    var c2 = countStatus_();
    if (c2.passed + c2.redeemed >= s.PRIZE_LIMIT) {
      try { FormApp.openById(s.FORM_ID).setAcceptingResponses(false)
        .setCustomClosedFormMessage('ขออภัย ของที่ระลึกหมดแล้ว ขอบคุณที่ร่วมงาน'); } catch (err) {}
    }
  } else {
    log.appendRow([d.time, d.code, d.name, d.license, "'" + d.phone, 'ไม่ผ่าน', reason, d.score, resp.getId(), d.line, d.hospital, d.specialty]);
  }
  SpreadsheetApp.flush();

  if (s.SEND_EMAIL && d.email) sendResultEmail_(s, d, reason);
}

/** JSON API สำหรับแอปสแกนด้วยกล้องมือถือ: ?action=ping|lookup|redeem&code=...&pin=...&lane=... */
/** เรียกจากหน้า Scanner (google.script.run) */
function scanApi(P) { return apiCore_(P || {}); }

function apiJson_(P) {
  var out = apiCore_(P);
  var js = JSON.stringify(out);
  if (P.callback && /^[A-Za-z_$][\w$]*$/.test(P.callback)) {
    return ContentService.createTextOutput(P.callback + '(' + js + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(js).setMimeType(ContentService.MimeType.JSON);
}

function apiCore_(P) {
  var out;
  try {
    var s = getSettings_();
    var pinOk = !!s.STAFF_PIN && String(P.pin || '').trim() === s.STAFF_PIN;
    if (!pinOk) {
      out = { ok: false, error: 'PIN ไม่ถูกต้อง' };
    } else if (P.action === 'ping') {
      var cnt = countStatus_();
      out = { ok: true, event: s.EVENT_NAME, passed: cnt.passed, redeemed: cnt.redeemed, limit: s.PRIZE_LIMIT };
    } else {
      var code = extractCode_(P.code || '');
      if (!code) out = { ok: false, error: 'ไม่พบรหัสบัตรใน QR' };
      else if (P.action === 'lookup') out = { ok: true, data: lookup(code) };
      else if (P.action === 'redeem') {
        // rid = รหัสคำขอจากหน้าสแกน: ถ้าเน็ตหลุดแล้วหน้าเว็บส่งซ้ำ จะได้ผลเดิม (ไม่ขึ้น "รับไปแล้ว" ผิด ๆ)
        var rid = String(P.rid || '').slice(0, 40), cache = CacheService.getScriptCache();
        var prev = rid ? cache.get('rid:' + rid) : null;
        if (prev) { out = JSON.parse(prev); }
        else {
          var force = String(P.force || '') === '1';
          var r = redeemByCode_(code, { force: force, lane: String(P.lane || ''), note: String(P.note || ''), name: String(P.name || '') });
          var state = r.ok ? (r.forced ? 'forced_now' : 'redeemed_now') : (r.state || 'notpassed');
          out = { ok: r.ok, state: state, msg: r.msg, code: code, name: r.name || '', score: r.score || '', at: r.at || '', lane: String(P.lane || ''), canForce: state === 'notpassed' };
          if (rid) cache.put('rid:' + rid, JSON.stringify(out), 900);
        }
      } else out = { ok: false, error: 'action ไม่ถูกต้อง' };
    }
  } catch (err) {
    out = { ok: false, error: String(err && err.message || err) };
  }
  return out;
}

/** เรียกจากหน้าเว็บ: ดูสถานะของรหัส */
function lookup(code) {
  code = normCode_(code);
  var s = getSettings_();
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
  var n = Math.max(0, sh.getLastRow() - 1);
  var codes = n ? sh.getRange(2, 1, n, 1).getValues() : [];
  var idx = -1;
  for (var i = 0; i < codes.length; i++) if (codes[i][0] === code) { idx = i; break; }
  var base = { event: s.EVENT_NAME, code: code };
  if (idx < 0) return merge_(base, { state: 'notfound' });

  var row = sh.getRange(idx + 2, 1, 1, CODE_HEADERS.length).getValues()[0];
  var status = row[C.STATUS - 1];
  if (status === ST_REDEEMED) return merge_(base, { state: 'redeemed', name: maskName_(row[C.NAME - 1]), at: fmt_(row[C.REDEEMED_AT - 1]) });
  if (status === ST_PASSED) return merge_(base, { state: 'passed', name: maskName_(row[C.NAME - 1]), score: row[C.SCORE - 1], at: fmt_(row[C.ANSWERED_AT - 1]) });

  // ยังไม่ได้สิทธิ์
  var now = new Date();
  var cnt = countStatus_();
  if (cnt.passed + cnt.redeemed >= s.PRIZE_LIMIT) return merge_(base, { state: 'soldout' });
  if (s.START_TIME && now < s.START_TIME) return merge_(base, { state: 'notyet', opensAt: fmt_(s.START_TIME) });
  if (s.END_TIME && now > s.END_TIME) return merge_(base, { state: 'closed' });

  var tmpl = PropertiesService.getScriptProperties().getProperty('PREFILL_TEMPLATE') || '';
  var formUrl = tmpl.replace('CODEPLACEHOLDER', encodeURIComponent(code));
  var lastFail = lastFailure_(code);
  return merge_(base, { state: 'open', formUrl: formUrl, lastFail: lastFail });
}

/** เรียกจากหน้าเว็บ: เจ้าหน้าที่ยืนยันการมอบของ */
function redeem(code, pin) {
  code = normCode_(code);
  var cache = CacheService.getScriptCache();
  var fails = Number(cache.get('pinfail') || 0);
  if (fails >= 10) return { ok: false, msg: 'ใส่ PIN ผิดหลายครั้ง ระบบล็อก 10 นาที' };

  var s = getSettings_();
  if (!s.STAFF_PIN || String(pin).trim() !== s.STAFF_PIN) {
    cache.put('pinfail', String(fails + 1), 600);
    return { ok: false, msg: 'PIN ไม่ถูกต้อง' };
  }

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
    var n = Math.max(0, sh.getLastRow() - 1);
    var codes = n ? sh.getRange(2, 1, n, 1).getValues() : [];
    for (var i = 0; i < codes.length; i++) {
      if (codes[i][0] !== code) continue;
      var r = i + 2;
      var st = sh.getRange(r, C.STATUS).getValue();
      if (st === ST_REDEEMED) return { ok: false, msg: 'บัตรนี้รับของไปแล้ว', data: lookup(code) };
      if (st !== ST_PASSED) return { ok: false, msg: 'บัตรนี้ยังไม่ได้ตอบผ่าน', data: lookup(code) };
      sh.getRange(r, C.STATUS).setValue(ST_REDEEMED);
      sh.getRange(r, C.REDEEMED_AT).setValue(new Date());
      queueThanks_(sh, r);
      SpreadsheetApp.flush();
      return { ok: true, msg: 'บันทึกการมอบของเรียบร้อย', data: lookup(code) };
    }
    return { ok: false, msg: 'ไม่พบรหัสบัตร' };
  } finally {
    lock.releaseLock();
  }
}

function lastFailure_(code) {
  var log = SpreadsheetApp.getActive().getSheetByName(SH_LOG);
  var n = log.getLastRow() - 1;
  if (n < 1) return '';
  var start = Math.max(2, log.getLastRow() - 500);
  var rows = log.getRange(start, 1, log.getLastRow() - start + 1, 7).getValues();
  for (var i = rows.length - 1; i >= 0; i--) {
    if (rows[i][1] === code) return rows[i][5] === 'ไม่ผ่าน' ? rows[i][6] : '';
  }
  return '';
}

function maskName_(name) {
  name = String(name || '').trim();
  var parts = name.split(/\s+/);
  if (parts.length < 2) return name;
  var last = parts[parts.length - 1];
  return parts.slice(0, -1).join(' ') + ' ' + last.charAt(0) + '•••';
}
function fmt_(d) { return d instanceof Date ? Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm') : String(d || ''); }
function merge_(a, b) { var o = {}; for (var k in a) o[k] = a[k]; for (var k2 in b) o[k2] = b[k2]; return o; }

function onEdit(e) {
  try {
    var r = e.range;
    var sh = r.getSheet();
    if (sh.getName().indexOf(SH_SCAN) === 0) { handleScan_(sh, r, e); return; }
    if (sh.getName() !== SH_REDEEM) return;
    if (r.getA1Notation() === 'B2') { sh.getRange('B12').setValue(false); sh.getRange('B13').setValue(''); return; }
    if (r.getA1Notation() !== 'B12' || e.value !== 'TRUE') return;
    var res = redeemByCode_(String(sh.getRange('B2').getValue()));
    sh.getRange('B13').setValue((res.ok ? '✅ ' : '❌ ') + res.msg);
    sh.getRange('B12').setValue(false);
  } catch (err) {
    try { e.range.getSheet().getRange('B13').setValue('❌ ' + err.message); } catch (e2) {}
  }
}

/**
 * ตัดสิทธิ์บัตร  opt = { force, lane, note, name }
 *   force = true → "ปล่อยผ่าน": มอบของแม้ยังไม่ได้ตอบผ่าน (บันทึกหมายเหตุ + ลงชีต "บันทึก" ว่าใคร/ช่องไหนปล่อย)
 */
function redeemByCode_(code, opt) {
  opt = opt || {};
  code = normCode_(code);
  if (!code) return { ok: false, state: 'notfound', msg: 'กรุณาพิมพ์รหัสบัตรในช่อง B2' };
  var lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
    ensureCodeCols_(sh);
    var n = Math.max(0, sh.getLastRow() - 1);
    if (!n) return { ok: false, state: 'notfound', msg: 'ไม่พบรหัสบัตร ' + code };
    var W = CODE_HEADERS.length;
    var data = sh.getRange(2, 1, n, W).getValues();          // อ่านครั้งเดียวทั้งตาราง (เร็วกว่าอ่านทีละช่อง)
    var i = -1;
    for (var k = 0; k < data.length; k++) if (data[k][0] === code) { i = k; break; }
    if (i < 0) return { ok: false, state: 'notfound', msg: 'ไม่พบรหัสบัตร ' + code };
    var row = i + 2, v = data[i];
    var st = v[C.STATUS - 1], name = String(v[C.NAME - 1] || '');
    if (st === ST_REDEEMED) return { ok: false, state: 'already', name: name, score: v[C.SCORE - 1], at: fmt_(v[C.REDEEMED_AT - 1]), msg: 'บัตร ' + code + ' รับของไปแล้วเมื่อ ' + fmt_(v[C.REDEEMED_AT - 1]) };
    var forced = false, now = new Date();
    if (st !== ST_PASSED) {
      if (!opt.force) return { ok: false, state: 'notpassed', name: name, msg: 'บัตร ' + code + ' ยังไม่ได้ตอบผ่าน (สถานะ: ' + st + ')' };
      forced = true;
      if (opt.name && !name) name = opt.name;
      v[C.NOTE - 1] = 'ปล่อยผ่าน (เดิม: ' + st + ')' + (opt.lane ? ' ช่อง ' + opt.lane : '') + (opt.note ? ' · ' + opt.note : '');
      var log = SpreadsheetApp.getActive().getSheetByName(SH_LOG);
      if (log) log.appendRow([now, code, name, '', '', 'ปล่อยผ่าน', v[C.NOTE - 1], '', '']);
    } else if (opt.lane) {
      v[C.NOTE - 1] = 'ช่อง ' + opt.lane;
    }
    v[C.STATUS - 1] = ST_REDEEMED;
    v[C.NAME - 1] = name;
    v[C.REDEEMED_AT - 1] = now;
    try {
      var s = getSettings_();
      var email = String(v[C.EMAIL - 1] || '').trim();
      if (s.THANKS_EMAIL && /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email) && !v[C.THANKS - 1]) v[C.THANKS - 1] = TH_PENDING;
    } catch (e) {}
    // เขียนกลับครั้งเดียว: คอลัมน์ B..L (สถานะ ชื่อ … อีเมลขอบคุณ)
    sh.getRange(row, C.STATUS, 1, C.THANKS - C.STATUS + 1).setValues([v.slice(C.STATUS - 1, C.THANKS)]);
    if (forced) sh.getRange(row, C.NOTE).setBackground('#ffe8b3');
    SpreadsheetApp.flush();
    return { ok: true, forced: forced, state: forced ? 'forced_now' : 'redeemed_now', name: name, score: v[C.SCORE - 1], at: fmt_(now),
             msg: (forced ? 'ปล่อยผ่าน – มอบของให้ ' : 'มอบของให้ ') + (name || '(ไม่มีชื่อ)') + ' (บัตร ' + code + ') เรียบร้อย ' + fmt_(now) };
  } finally {
    lock.releaseLock();
  }
}

/** เพิ่มหัวคอลัมน์ K/L ในชีต "รหัส" ถ้ายังไม่มี (รองรับชีตที่สร้างด้วยโค้ดเวอร์ชันก่อน) */
function ensureCodeCols_(sh) {
  if (!sh) return;
  if (sh.getMaxColumns() < CODE_HEADERS.length) sh.insertColumnsAfter(sh.getMaxColumns(), CODE_HEADERS.length - sh.getMaxColumns());
  var head = sh.getRange(1, 1, 1, CODE_HEADERS.length).getValues()[0];
  var changed = false;
  for (var i = 0; i < CODE_HEADERS.length; i++) if (!head[i]) { head[i] = CODE_HEADERS[i]; changed = true; }
  if (changed) sh.getRange(1, 1, 1, CODE_HEADERS.length).setValues([head]).setFontWeight('bold').setBackground('#1f4e79').setFontColor('#ffffff');
}

// ---------- อีเมลขอบคุณหลังรับของ (ส่งเป็นคิว ไม่ถ่วงความเร็วจุดสแกน) ----------
function queueThanks_(sh, row) {
  try {
    var s = getSettings_();
    if (!s.THANKS_EMAIL) return;
    var email = String(sh.getRange(row, C.EMAIL).getValue() || '').trim();
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return;
    var cell = sh.getRange(row, C.THANKS);
    if (!cell.getValue()) cell.setValue(TH_PENDING);
  } catch (err) { console.error(err); }
}

/** ส่งอีเมลขอบคุณที่ค้างอยู่ (เรียกจากเมนู หรือ time trigger ทุก 5 นาที) – ส่งเท่าที่โควตาวันนี้เหลือ */
function sendPendingThanks() {
  var s = getSettings_();
  var sh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
  ensureCodeCols_(sh);
  var n = Math.max(0, sh.getLastRow() - 1);
  if (!n) return { sent: 0, left: 0, quota: MailApp.getRemainingDailyQuota() };
  var vals = sh.getRange(2, 1, n, CODE_HEADERS.length).getValues();
  var sent = 0, left = 0;
  for (var i = 0; i < vals.length; i++) {
    var r = vals[i];
    if (r[C.STATUS - 1] !== ST_REDEEMED || r[C.THANKS - 1] !== TH_PENDING) continue;
    var email = String(r[C.EMAIL - 1] || '').trim();
    if (!email) { sh.getRange(i + 2, C.THANKS).setValue('ไม่มีอีเมล'); continue; }
    if (MailApp.getRemainingDailyQuota() < 1) { left++; continue; }
    try {
      var name = String(r[C.NAME - 1] || '').trim();
      var subject = '🎁 ' + s.EVENT_NAME + ' – ขอบคุณที่ร่วมกิจกรรม (ยืนยันการรับของที่ระลึก)';
      var body = 'เรียน ' + (name ? 'คุณ' + name : 'ท่านผู้เข้าร่วมประชุม') + '\n\n' +
        (s.THANKS_TEXT ? s.THANKS_TEXT + '\n\n' : '') +
        'อีเมลนี้เป็นการยืนยันว่าท่านได้รับของที่ระลึกเรียบร้อยแล้ว\n' +
        '  บัตรหมายเลข: ' + r[C.CODE - 1] + '\n' +
        '  เวลารับของ: ' + fmt_(r[C.REDEEMED_AT - 1]) + '\n\n' +
        'ขอบคุณที่ร่วมงาน\n' + s.EVENT_NAME + '\nTRB Chemedica (Thailand) Ltd.';
      MailApp.sendEmail({ to: email, subject: subject, body: body, name: s.EVENT_NAME });
      sh.getRange(i + 2, C.THANKS).setValue('ส่งแล้ว ' + fmt_(new Date()));
      sent++;
    } catch (err) {
      sh.getRange(i + 2, C.THANKS).setValue('ผิดพลาด: ' + String(err.message || err).slice(0, 60));
    }
  }
  return { sent: sent, left: left, quota: MailApp.getRemainingDailyQuota() };
}

function sendPendingThanksMenu() {
  var r = sendPendingThanks();
  SpreadsheetApp.getUi().alert('ส่งอีเมลขอบคุณแล้ว ' + r.sent + ' ฉบับ' + (r.left ? '\nค้างอยู่ ' + r.left + ' ฉบับ (โควตาวันนี้หมด – เลือกเมนูนี้อีกครั้งพรุ่งนี้ หรือเปิดส่งอัตโนมัติ)' : '') +
    '\nโควตาอีเมลคงเหลือวันนี้: ' + r.quota);
}

function installThanksTrigger() {
  removeThanksTrigger(true);
  ScriptApp.newTrigger('sendPendingThanks').timeBased().everyMinutes(5).create();
  SpreadsheetApp.getUi().alert('เปิดส่งอีเมลขอบคุณอัตโนมัติแล้ว (ทุก 5 นาที ส่งเท่าที่โควตาวันนี้เหลือ ที่เหลือจะส่งต่อวันถัดไปเอง)\nอย่าลืมตั้ง THANKS_EMAIL = TRUE ในชีต "ตั้งค่า"');
}
function removeThanksTrigger(silent) {
  var n = 0;
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'sendPendingThanks') { ScriptApp.deleteTrigger(t); n++; } });
  if (!silent) SpreadsheetApp.getUi().alert(n ? 'ปิดส่งอีเมลอัตโนมัติแล้ว' : 'ไม่ได้เปิดส่งอัตโนมัติอยู่');
}

/** ดึงรหัสบัตรออกจากสิ่งที่สแกนได้ (ลิงก์ฟอร์มที่เติมรหัสไว้ / ลิงก์เว็บแอป / รหัสล้วน) */
function extractCode_(raw) {
  raw = String(raw || '').trim();
  if (!raw) return '';
  var m = raw.match(/(?:^|[?&])(?:entry\.\d+|code|k)=([^&#\s]+)/i);
  if (m) { try { raw = decodeURIComponent(m[1].replace(/\+/g, ' ')); } catch (err) { raw = m[1]; } }
  raw = normCode_(raw);
  var m2 = raw.match(/([A-Z]+\d{3})-?([A-Z0-9]{4})/);
  return m2 ? m2[1] + '-' + m2[2] : raw;
}

function handleScan_(sh, r, e) {
  if (r.getA1Notation() !== SCAN_INPUT) return;
  var raw = e.value != null ? e.value : r.getValue();
  if (!raw) return;
  var force = /!\s*$/.test(String(raw));          // พิมพ์รหัสแล้วต่อท้ายด้วย ! = ปล่อยผ่าน (มอบของแม้ยังไม่ตอบผ่าน)
  var code = extractCode_(String(raw).replace(/!\s*$/, ''));
  var lane = String(sh.getName() === SH_SCAN ? '1' : sh.getName().replace(SH_SCAN + '-', ''));
  var res = redeemByCode_(code, { force: force, lane: 'N' + lane });
  var name = '', color, text;
  var codesSh = SpreadsheetApp.getActive().getSheetByName(SH_CODES);
  var row = findCodeRow_(codesSh, code);
  if (row) name = String(codesSh.getRange(row, C.NAME).getValue() || '');
  if (res.ok && res.forced) { color = '#c6efce'; text = '✅ ปล่อยผ่าน'; }
  else if (res.ok) { color = '#c6efce'; text = '✅ มอบของได้'; }
  else if (/รับของไปแล้ว/.test(res.msg)) { color = '#ffc7ce'; text = '❌ รับไปแล้ว'; }
  else if (/ไม่พบ/.test(res.msg)) { color = '#ffc7ce'; text = '❌ ไม่พบรหัส'; }
  else { color = '#ffe8b3'; text = '⚠ ยังไม่ผ่าน'; }
  sh.getRange(SCAN_RESULT).setValue(text + '  ' + code).setBackground(color);
  sh.getRange(SCAN_DETAIL).setValue((name ? name + ' · ' : '') + res.msg + (text === '⚠ ยังไม่ผ่าน' ? '   (ถ้าต้องปล่อยผ่าน: พิมพ์ ' + code + '! แล้ว Enter)' : ''));
  // ประวัติ (แถวใหม่บนสุด, เก็บไม่เกิน SCAN_HIST_MAX)
  sh.insertRowAfter(SCAN_HIST_HEAD);
  sh.getRange(SCAN_HIST_HEAD + 1, 1, 1, 4).setValues([[new Date(), code, text, name]])
    .setBackground(color).setFontColor('#1d2733').setFontWeight('normal').setNumberFormat('@');
  sh.getRange(SCAN_HIST_HEAD + 1, 1).setNumberFormat('dd/mm/yyyy hh:mm:ss');
  var last = sh.getLastRow();
  if (last > SCAN_HIST_HEAD + SCAN_HIST_MAX) sh.deleteRows(SCAN_HIST_HEAD + SCAN_HIST_MAX + 1, last - (SCAN_HIST_HEAD + SCAN_HIST_MAX));
  // เคลียร์ช่องและกลับไปรอสแกนใบต่อไป
  var inp = sh.getRange(SCAN_INPUT);
  inp.setValue('');
  inp.activate();
}

function findCodeRow_(sh, code) {
  if (!sh || !code) return 0;
  var n = Math.max(0, sh.getLastRow() - 1);
  if (!n) return 0;
  var codes = sh.getRange(2, 1, n, 1).getValues();
  for (var i = 0; i < codes.length; i++) if (codes[i][0] === code) return i + 2;
  return 0;
}

/** ข้อความหน้ายืนยันหลังส่งคำตอบ – ต่อท้ายด้วย REDEEM_NOTE จากชีต "ตั้งค่า" */

  var msg = 'ส่งคำตอบเรียบร้อย ✅\n\n';
  if (s.REDEEM_NOTE) msg += s.REDEEM_NOTE + '\n\n';

  try { graded = (JSON.parse(PropertiesService.getScriptProperties().getProperty('ITEM_IDS') || '{}').questions || []).some(function (q) { return q.answers ? q.answers.length > 0 : !!q.answer; }); } catch (e) {}
  msg += 'กรุณานำ "บัตร QR" ใบนี้มาแสดงที่จุดแลกของ เจ้าหน้าที่จะสแกนบัตรเพื่อยืนยันสิทธิ์' + (graded ? '\n(หากตอบไม่ผ่าน สแกน QR แล้วตอบใหม่ได้)' : '');
  return msg;
}
